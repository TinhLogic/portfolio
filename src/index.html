ZERO-ALLOCATION & STANDARDS

Chuẩn mực kiến trúc mã nguồn

Ứng dụng System.IO.Pipelines, ArrayPool, và RFC 9457 Problem Details nhằm triệt tiêu Garbage Collection pauses trong các dịch vụ phụ tải cao.

HighThroughputSocketHandler.cs 
// C# 12 / .NET 8: Zero-Allocation Request Stream Buffer Pipeline
public sealed class HighThroughputSocketHandler : IAsyncDisposable
{
    private readonly PipeReader _reader;
    private readonly ILogger<HighThroughputSocketHandler> _logger;

    public async ValueTask HandleStreamAsync(CancellationToken ct = default)
    {
        while (!ct.IsCancellationRequested)
        {
            ReadResult result = await _reader.ReadAsync(ct);
            ReadOnlySequence<byte> buffer = result.Buffer;

            while (TryProcessMessage(ref buffer, out var message))
            {
                await DispatchCommandAsync(message, ct);
            }

            _reader.AdvanceTo(buffer.Start, buffer.End);
            if (result.IsCompleted) break;
        }
    }
}
